```typescript
/**
 * Directories to filter from directory browser.
 * These appear as disabled (greyed out) with explanatory hint.
 */
const FILTERED_DIRS = new Set([
  // System directories
  'tmp', 'var', 'private', 'System', 'Volumes',
  // Common caches
  'node_modules', '.git', '.svn', '.hg',
  'Library', // On macOS, too large/not useful
  '__pycache__', '.pytest_cache', '.mypy_cache',
  'target', // Rust/Maven
  'build', 'dist', '.next', '.nuxt',
  // Package manager caches
  '.npm', '.yarn', '.pnpm', '.bun',
]);

/**
 * Default patterns for low-value dotfiles (caches, temp, history).
 * These are shown in "Other files" section with annotation.
 */
const DEFAULT_LOW_VALUE_PATTERNS = [
  // Exact matches
  '.DS_Store', '.localized', 'Thumbs.db', 'desktop.ini',
  '.CFUserTextEncoding',
  // History files
  '.zsh_history', '.bash_history', '.node_repl_history', '.python_history',
  '.lesshst', '.wget-hsts',
  // Sessions
  '.zsh_sessions', '.bash_sessions',
  // Caches
  '.cache', '.tmp',
];

/**
 * Pattern suffixes for low-value files.
 */
const LOW_VALUE_SUFFIXES = ['_history', '.log', '.bak', '.swp', '.swo'];

/**
 * Check if a filename matches low-value patterns.
 * Accepts optional custom patterns from config.
 */
function isLowValueFile(
  name: string,
  customPatterns?: { lowValue?: string[]; highValue?: string[] }
): boolean {
  // High-value overrides take priority
  if (customPatterns?.highValue?.includes(name)) return false;

  // Check custom low-value patterns
  if (customPatterns?.lowValue?.includes(name)) return true;

  // Check default patterns
  if (DEFAULT_LOW_VALUE_PATTERNS.includes(name)) return true;

  return LOW_VALUE_SUFFIXES.some(suffix => name.endsWith(suffix));
}
```

3. In `src/init.ts`, update the selection display to implement collapse behavior:

Since @clack/prompts multiselect doesn't support visual separators or collapsible sections, implement a two-phase approach:
   - First show valuable files in main multiselect
   - If >5 low-value files, show a "Show N more files..." option at the end
   - If user selects it, show a second multiselect with just the low-value files
   - If <=5 low-value files, include them directly (dimmed) in the main list

```typescript
// Separate valuable from low-value
const valuable = available.filter(df => !df.isLowValue);
const lowValue = available.filter(df => df.isLowValue);

// Build options for main selection
const selectableItems = [];

// Add valuable files first
for (const df of valuable) {
  selectableItems.push({
    value: df.name,
    label: df.name,
    hint: df.warning ?? `-> ${df.suggested}`,
  });
}

// Handle low-value files based on count
const COLLAPSE_THRESHOLD = 5;

if (lowValue.length > COLLAPSE_THRESHOLD) {
  // Add a "show more" option
  selectableItems.push({
    value: '__show_more__',
    label: pc.dim(`Show ${lowValue.length} more files...`),
    hint: pc.dim('cache, history, temp files'),
  });
} else {
  // Add low-value files directly (dimmed)
  for (const df of lowValue) {
    selectableItems.push({
      value: df.name,
      label: pc.dim(df.name),
      hint: pc.dim(df.annotation ?? 'may not be worth tracking'),
    });
  }
}

// After main selection, check if user selected __show_more__
const selected = await p.multiselect({ ... });

if (selected.includes('__show_more__')) {
  // Remove the marker and show second selection for low-value files
  const mainSelected = selected.filter(s => s !== '__show_more__');

  p.log.info('These files are typically caches, history, or temp files:');

  const lowValueOptions = lowValue.map(df => ({
    value: df.name,
    label: df.name,
    hint: df.annotation ?? 'may not be worth tracking',
  }));

  const lowValueSelected = await p.multiselect({
    message: 'Select any you want to track anyway:',
    options: lowValueOptions,
    required: false,
  });

  // Combine selections
  finalSelected = [...mainSelected, ...(p.isCancel(lowValueSelected) ? [] : lowValueSelected)];
} else {
  finalSelected = selected;
}
```

This achieves the spirit of "collapsed if >5, expanded if <=5" within @clack/prompts limitations.
