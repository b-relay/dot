Requirements:

   - File sink: JSON Lines format for machine parsing, writes to `.dot-operations.jsonl` in dotfiles directory
2. Export `configureLogging(dotfilesPath: string)` - must be called once at startup
3. Export `getLogger(category: string[])` - returns logger for a subsystem

Use the pattern from RESEARCH.md:
```typescript
import { configure, getConsoleSink, getLogger as getLogTapeLogger } from "@logtape/logtape";
import { getFileSink } from "@logtape/file";

let configured = false;

export async function configureLogging(dotfilesPath: string): Promise<void> {
  if (configured) return;

  await configure({
    sinks: {
      console: getConsoleSink(),
      file: getFileSink(`${dotfilesPath}/.dot-operations.jsonl`, {
        formatter: (record) => JSON.stringify({
          timestamp: record.timestamp.toISOString(),
          level: record.level,
          category: record.category,
          message: record.message,
          ...record.properties
        }) + "\n"
      })
    },
    loggers: [
      { category: ["dot"], lowestLevel: "info", sinks: ["console", "file"] }
    ]
  });

  configured = true;
}

export function getLogger(category: string[]): ReturnType<typeof getLogTapeLogger> {
  return getLogTapeLogger(["dot", ...category]);
}
```

Note: The file sink writes to dotfiles repo so logs travel with the repo. Use "info" as default level (debug is too verbose for normal use).
