Create install.sh at repository root (not in dot/):

```bash
#!/usr/bin/env bash
set -euo pipefail

# dot CLI installer
# Usage: curl -fsSL https://raw.githubusercontent.com/user/dotfiles/main/install.sh | bash

VERSION="${DOT_VERSION:-latest}"
INSTALL_DIR=""

# Colors
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[0;33m'
NC='\033[0m'

info() { echo -e "${GREEN}==>${NC} $1"; }
warn() { echo -e "${YELLOW}Warning:${NC} $1"; }
error() { echo -e "${RED}Error:${NC} $1" >&2; exit 1; }

# Detect architecture
detect_arch() {
  local arch=$(uname -m)
  case "$arch" in
    x86_64) echo "x64" ;;
    arm64|aarch64) echo "arm64" ;;
    *) error "Unsupported architecture: $arch" ;;
  esac
}

# Detect OS
detect_os() {
  local os=$(uname -s)
  case "$os" in
    Darwin) echo "darwin" ;;
    Linux) echo "linux" ;;
    *) error "Unsupported OS: $os" ;;
  esac
}

# Prompt for install location
choose_install_dir() {
  echo ""
  echo "Where would you like to install dot?"
  echo "  1) ~/.local/bin (recommended, user-local)"
  echo "  2) /usr/local/bin (system-wide, requires sudo)"
  echo ""
  read -p "Choice [1]: " choice

  case "${choice:-1}" in
    1) INSTALL_DIR="$HOME/.local/bin" ;;
    2) INSTALL_DIR="/usr/local/bin" ;;
    *) error "Invalid choice" ;;
  esac
}

# Check if directory is in PATH
check_path() {
  if [[ ":$PATH:" != *":$INSTALL_DIR:"* ]]; then
    warn "$INSTALL_DIR is not in your PATH"
    echo "Add this to your shell config:"
    echo "  export PATH=\"$INSTALL_DIR:\$PATH\""
  fi
}

# Download and install
install() {
  local arch=$(detect_arch)
  local os=$(detect_os)
  local binary_name="dot-${os}-${arch}"

  # For now, build from source since we don't have releases yet
  # TODO: Replace with GitHub releases download when available
  info "Building dot from source..."

  if ! command -v bun &> /dev/null; then
    error "Bun is required to build from source. Install it first: curl -fsSL https://bun.sh/install | bash"
  fi

  # Clone or use existing repo
  local tmp_dir=$(mktemp -d)
  trap "rm -rf $tmp_dir" EXIT

  info "Cloning dot repository..."
  git clone --depth 1 https://github.com/brendonv/dotfiles.git "$tmp_dir" 2>/dev/null || \
    git clone --depth 1 git@github.com:brendonv/dotfiles.git "$tmp_dir"

  info "Building binary..."
  cd "$tmp_dir/dot"
  bun install
  bun run build:release

  # Install binary
  info "Installing to $INSTALL_DIR..."
  mkdir -p "$INSTALL_DIR"

  if [[ "$INSTALL_DIR" == "/usr/local/bin" ]]; then
    sudo cp dot "$INSTALL_DIR/dot"
    sudo chmod +x "$INSTALL_DIR/dot"
  else
    cp dot "$INSTALL_DIR/dot"
    chmod +x "$INSTALL_DIR/dot"
  fi

  check_path

  echo ""
  info "dot installed successfully!"
  echo ""
  echo "Get started:"
  echo "  dot init    # Set up your dotfiles"
  echo "  dot help    # See all commands"
}

# Main
main() {
  echo "dot CLI installer"
  echo ""

  choose_install_dir
  install
}

main "$@"
```

- Script builds from source since we don't have releases yet

- Handles both ~/.local/bin and /usr/local/bin
- Warns if install dir not in PATH
